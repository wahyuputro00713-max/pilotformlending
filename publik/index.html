function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var doc = SpreadsheetApp.getActiveSpreadsheet();
    var data = JSON.parse(e.postData.contents);
    var targetSheetName = data.pulau || "Jawa 1";
    var sheet = doc.getSheetByName(targetSheetName);
    
    if (!sheet) {
      sheet = doc.insertSheet(targetSheetName);
      var headers = ["Tanggal", "Pulau", "Regional", "Area", "Point", "NIK BP", "Nama BP", "Nama Calon Mitra", "Nomer HP", "POI", "NIK KTP Mitra", "Akan di Proses", "Pilih Pulau POI", "Nama Lokasi POI"];
      sheet.appendRow(headers);
      sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#0f172a").setFontColor("#ffffff");
    }
    
    var rowData = [
      data.tanggal, data.pulau, data.regional, data.area, data.point,
      data.nik_bp, data.nama_bp, data.nama_calon_mitra, data.nomer_hp, data.poi,
      data.nik_ktp_mitra, data.akan_di_proses, data.pilih_pulau_poi, data.nama_lokasi_poi
    ];
    sheet.appendRow(rowData);
    return ContentService.createTextOutput(JSON.stringify({"result": "success", "sheet": targetSheetName})).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({"result": "error", "error": err.toString()})).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function doGet(e) {
  try {
    var doc = SpreadsheetApp.getActiveSpreadsheet();
    var masterSheet = doc.getSheetByName("Data Pulau");
    if (!masterSheet) return ContentService.createTextOutput(JSON.stringify({"status": "error"})).setMimeType(ContentService.MimeType.JSON);
    var values = masterSheet.getRange(2, 1, masterSheet.getLastRow() - 1, 4).getValues();
    var listData = values.map(function(v) { return { pulau: String(v[0]).trim(), regional: String(v[1]).trim(), area: String(v[2]).trim(), point: String(v[3]).trim() }; });
    return ContentService.createTextOutput(JSON.stringify({"status": "success", "data": listData})).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({"status": "error", "error": err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}
